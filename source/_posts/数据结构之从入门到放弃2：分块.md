## $Description$

已知一个数列，你需要进行下面两种操作：

1.将某区间每一个数加上$x$

2.求出某区间每一个数的和

## $Solution$

~~似 曾 相 识~~

分块，也是一种数据结构，它相对于树状数组和线段树来说，通用直观，更好理解

因为好理解，我就不找图了~~才不是懒呢~~

### 基础思想：

我们把数列a分成若干个长度为$\sqrt{N}$的段（最后一段可以不足）

第i段的左端点为(i - 1)  *$\sqrt{N}$ + 1，右端点为i*  $\sqrt{N}$ (最后一个为N)

开两个数组sum和add，sum表示这一段的和，add表示这一段增加的值

左端点，右端点，sum都可以预处理出来

开个数组L[x]表示第x段的左端点，R[x]表示第x段的右端点

再开一个数组pos[x]表示x在第几段  
```
void ycl(){
	int u = sqrt(n);
	for(int i = 1; i <= u; ++i){
		L[i] = (i - 1) * u + 1;
		R[i] = i * u;
	}
	if(R[u] != u){
		++u;
		L[u] = R[u - 1] + 1; R[u] = N;
	} //如果n不是完全平方数，最后肯定会剩下一小段没算
	for(int i = 1; i <= u; ++i){
		for(int j = L[i]; j <= R[i]; ++j){
			sum[i] += a[j];
			pos[j] = i;
		}
	} 
}
```

### 区间修改

假设要修改的左端点是l，右端点是r，加上d

简单地说，对于不足一段的，直接暴力加a和sum，中间的就只加add

意思是：找到左端点所在的段数p，右端点所在的段数q

在l到L[p]，R[q]到r之间的区域，直接暴力循环，给每个a都加上d,sum[p] += d  *(R[p] - l + 1), sum[q] += d*  (r - L[q] + 1)

对于 p + 1 到 q - 1 之间的段落，给每个add加上d，代表这一段的所有数字都要加上d

注意特判：如果p==q，那么直接从l到r暴力就行了

容易看出，该操作是O($\sqrt{N}$)的  
```
void change(int l, int r, int d){
	int p = pos[l], q = pos[r];
	if(p == q){
		for(int i = l; i <= r; ++i) a[i] += d;
		sum[p] += d * (r - l + 1);
		return;
	}
	for(int i = p + 1; i <= q - 1; ++i) add[i] += d;
	for(int i = l; i <= R[p]; ++i) a[i] += d;
	sum[p] += d * (R[p] - l + 1)
	for(int i = L[q]; i <= r; ++i) a[i] += d;
	sum[q] += d * (r - L[q] + 1)
}
```

### 区间查询

和区间修改一模一样啊

不足一段的直接暴力加，中间的就加上sum[i] + add[i] * (R[i] - L[i] + 1)

记得特判

~~是不是一瞬间觉得分块简直太友好了~~

```
int query(int l, int r){
	int ans = 0;
	int p = pos[l], q = pos[r];
	if(p == q){
		for(int i = l; i <= r; ++i) ans += a[i];
		return ans;
	}
	for(int i = p + 1; i <= q - 1; ++i) ans += sum[i] + add[i] * (R[i] - L[i] + 1);
	for(int i = l; i <= R[p]; ++i) ans += a[i];
	ans += add[p] * (R[p] - l + 1);
	for(int i = L[q]; i <= r; ++i) ans += a[i];
	ans += add[q] * (r - L[q] + 1);
	return ans;
}
```

## $Code$

```
#include <bits/stdc++.h>
#define N 100005
#define int long long
using namespace std;

int n, m, op, l, r, x, y, d;
int a[N], L[N], R[N], pos[N], sum[N], add[N];

void ycl(){
	int u = sqrt(n);
	for(int i = 1; i <= u; ++i){
		L[i] = (i - 1) * u + 1;
		R[i] = i * u;
	}
	if(R[u] != u){
		++u;
		L[u] = R[u - 1] + 1; R[u] = N;
	} 
	for(int i = 1; i <= u; ++i){
		for(int j = L[i]; j <= R[i]; ++j){
			sum[i] += a[j];
			pos[j] = i;
		}
	} 
}

void change(int l, int r, int d){
	int p = pos[l], q = pos[r];
	if(p == q){
		for(int i = l; i <= r; ++i) a[i] += d;
		sum[p] += d * (r - l + 1);
		return;
	}
	for(int i = p + 1; i <= q - 1; ++i) add[i] += d;
	for(int i = l; i <= R[p]; ++i) a[i] += d;
	sum[p] += d * (R[p] - l + 1)
	for(int i = L[q]; i <= r; ++i) a[i] += d;
	sum[q] += d * (r - L[q] + 1)
}

int query(int l, int r){
	int ans = 0;
	int p = pos[l], q = pos[r];
	if(p == q){
		for(int i = l; i <= r; ++i) ans += a[i];
		return ans;
	}
	for(int i = p + 1; i <= q - 1; ++i) ans += sum[i] + add[i] * (R[i] - L[i] + 1);
	for(int i = l; i <= R[p]; ++i) ans += a[i];
	ans += add[p] * (R[p] - l + 1);
	for(int i = L[q]; i <= r; ++i) ans += a[i];
	ans += add[q] * (r - L[q] + 1);
	return ans;
}
 
signed main(){
	scanf("%d%d", &n, &m);
	for(int i = 1; i <= n; ++i) scanf("%d", &a[i]);
	ycl();
	for(int i = 1; i <= m; ++i){
		scanf("%d", &op);
		if(op == 1){
			scanf("%d%d%d", &l, &r, &d);
			change(l, r, d);
		}
		else{
			scanf("%d%d", &l, &r);
			printf("%d\n", query(l, r));
		}
	}	
	return 0;
}//跟线段树行数一模一样QAQAQAQ
```
